/* ========================================================= BACKEND — Controle de Gastos da Casa Cole este código no Apps Script vinculado à sua planilha (Extensões > Apps Script, dentro do Google Sheets) ========================================================= */ // 1) Cole aqui o ID da sua planilha (está na URL, entre /d/ e /edit) const SHEET_ID = 'COLE_AQUI_O_ID_DA_PLANILHA'; // 2) Nome da aba onde os gastos serão salvos const SHEET_NAME = 'Gastos'; // 3) (Opcional) ID de uma pasta do Google Drive para guardar as fotos das notas. // Deixe '' em branco se não quiser salvar fotos por enquanto. const DRIVE_FOLDER_ID = ''; function doGet(e) { const sheet = getSheet_(); const data = sheet.getDataRange().getValues(); const headers = data.shift(); const gastos = data .filter(row => row[0] !== '') .map(row => { const obj = {}; headers.forEach((h, i) => { let val = row[i]; if (h === 'Data') { if (Object.prototype.toString.call(val) === '[object Date]') { val = Utilities.formatDate(val, Session.getScriptTimeZone(), 'yyyy-MM-dd'); } else if (typeof val === 'string' && val.indexOf('T') > -1) { val = val.split('T')[0]; } } else if (val instanceof Date) { val = Utilities.formatDate(val, Session.getScriptTimeZone(), 'yyyy-MM-dd'); } obj[h] = val; }); return obj; }); return resposta_({ gastos }); } function doPost(e) { const body = JSON.parse(e.postData.contents); const sheet = getSheet_(); if (body.action === 'add') { let comprovanteUrl = ''; if (body.foto) comprovanteUrl = salvarFoto_(body.foto, body.fotoNome); sheet.appendRow([ body.id || Date.now().toString(), body.data, body.descricao, Number(body.valor), body.quem, comprovanteUrl, new Date() ]); return resposta_({ ok: true }); } if (body.action === 'edit') { const linha = acharLinha_(sheet, body.id); if (linha > -1) { let comprovanteUrl = null; if (body.foto) comprovanteUrl = salvarFoto_(body.foto, body.fotoNome || 'comprovante.jpg'); sheet.getRange(linha, 2).setValue(body.data); sheet.getRange(linha, 3).setValue(body.descricao); sheet.getRange(linha, 4).setValue(Number(body.valor)); sheet.getRange(linha, 5).setValue(body.quem); if (comprovanteUrl) sheet.getRange(linha, 6).setValue(comprovanteUrl); } return resposta_({ ok: true }); } if (body.action === 'delete') { const linha = acharLinha_(sheet, body.id); if (linha > -1) sheet.deleteRow(linha); return resposta_({ ok: true }); } return resposta_({ ok: false, erro: 'Ação inválida' }); } /* ---------- Funções auxiliares ---------- */ function getSheet_() { const ss = SpreadsheetApp.openById(SHEET_ID); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); sheet.appendRow(['ID', 'Data', 'Descricao', 'Valor', 'Quem', 'Comprovante', 'Timestamp']); } return sheet; } function acharLinha_(sheet, id) { const ids = sheet.getRange(2, 1, Math.max(sheet.getLastRow() - 1, 0), 1).getValues(); for (let i = 0; i < ids.length; i++) { if (String(ids[i][0]) === String(id)) return i + 2; } return -1; } function salvarFoto_(base64, nome) { if (!DRIVE_FOLDER_ID) return ''; try { const folder = DriveApp.getFolderById(DRIVE_FOLDER_ID); const partes = base64.split(','); const tipo = partes[0].match(/data:(.*);base64/)[1]; const blob = Utilities.newBlob(Utilities.base64Decode(partes[1]), tipo, nome || 'comprovante.jpg'); const file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); return file.getUrl(); } catch (err) { return ''; } } function resposta_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
